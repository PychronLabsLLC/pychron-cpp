# pychron::persistence — the DVC store (DVC schema spec, section 12) on
# TinyORM/QtSql. Depends on core only. TinyORM and Qt are PRIVATE: the public
# headers under include/pychron/persistence/ are std-only.
if(NOT PYCHRON_PERSISTENCE_ENABLED)
  return()
endif()

# Embed migrations/<dialect>/*.sql as byte arrays so the library carries its
# schema; schema_version records each file's SHA-256 (section 11.4).
file(GLOB PYCHRON_MIGRATION_FILES CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/migrations/pg/*.sql"
  "${CMAKE_CURRENT_SOURCE_DIR}/migrations/sqlite/*.sql")
list(SORT PYCHRON_MIGRATION_FILES)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PYCHRON_MIGRATION_FILES})
set(_embedded "// Generated by libs/persistence/CMakeLists.txt. Do not edit.\n#include \"migrate.hpp\"\n\nnamespace pychron::persistence::detail {\nnamespace {\n")
set(_table "")
set(_n 0)
foreach(path IN LISTS PYCHRON_MIGRATION_FILES)
  get_filename_component(_file "${path}" NAME)
  get_filename_component(_dir "${path}" DIRECTORY)
  get_filename_component(_dialect "${_dir}" NAME)
  file(READ "${path}" _hex HEX)
  string(LENGTH "${_hex}" _len)
  math(EXPR _size "${_len} / 2")
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _bytes "${_hex}")
  string(APPEND _embedded "const unsigned char kMigration${_n}[] = {${_bytes}};\n")
  string(APPEND _table "    {\"${_dialect}\", \"${_file}\", kMigration${_n}, ${_size}},\n")
  math(EXPR _n "${_n} + 1")
endforeach()
string(APPEND _embedded "}  // namespace\n\nconst EmbeddedMigration kEmbeddedMigrations[] = {\n${_table}};\nconst std::size_t kEmbeddedMigrationCount = ${_n};\n\n}  // namespace pychron::persistence::detail\n")
set(_embedded_cpp "${CMAKE_CURRENT_BINARY_DIR}/migrations_embedded.cpp")
file(CONFIGURE OUTPUT "${_embedded_cpp}" CONTENT "${_embedded}" @ONLY)

add_library(pychron_persistence STATIC
  src/blob.cpp
  src/change.cpp
  src/ids.cpp
  src/ingest.cpp
  src/migrate.cpp
  src/model.cpp
  src/payloads.cpp
  src/store.cpp
  src/unit_of_work.cpp
  src/sql/errors.cpp
  src/tiny/db.cpp
  "${_embedded_cpp}"
)
add_library(pychron::persistence ALIAS pychron_persistence)

target_include_directories(pychron_persistence
  PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
  PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/src)

target_compile_features(pychron_persistence PUBLIC cxx_std_20)
target_link_libraries(pychron_persistence PUBLIC pychron::core)
target_link_libraries(pychron_persistence PRIVATE TinyOrm::TinyOrm Qt6::Core Qt6::Sql)
# Qt keyword macros (signals, slots, emit) would rewrite ordinary identifiers.
target_compile_definitions(pychron_persistence PRIVATE QT_NO_KEYWORDS)

pychron_set_warnings(pychron_persistence)
